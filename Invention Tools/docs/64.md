def copy_sheet(src_wb, src_name, dest_wb, dest_name):
    ws_src = src_wb[src_name]
    ws_dst = dest_wb.create_sheet(dest_name)
    for mc in ws_src.merged_cells.ranges:
        ws_dst.merge_cells(str(mc))
    from openpyxl.cell.cell import MergedCell
    for row in ws_src.iter_rows():
        for cell in row:
            if isinstance(cell, MergedCell):
                continue
            new_cell = ws_dst.cell(row=cell.row, column=cell.column)
            new_cell.value = cell.value

                new_cell.font = copy(cell.font)
